## example code
    #include <stdio.h>

    // 1. 指针作为函数参数：可以修改函数外部的变量
    // 这个函数的作用是交换两个变量的值
    void swap(int *a, int *b) {
        int temp = *a; // *a 就是获取指针 a 所指向地址的值
        *a = *b;       // *b 就是获取指针 b 所指向地址的值
        *b = temp;
    }

    int main() {
        // --- 第一部分：指针与基本变量 ---
        int x = 10;
        int y = 20;
        
        printf("交换前: x = %d, y = %d\n", x, y);
        
        // 调用函数时，传入的是变量的地址 (&x, &y)
        swap(&x, &y);
        
        printf("交换后: x = %d, y = %d\n", x, y);


        // --- 第二部分：指针与数组 ---
        int arr[] = {100, 200, 300, 400, 500};
        int *ptr = arr; // 数组名 arr 本身就代表了数组第一个元素的地址
        
        printf("\n使用指针遍历数组: ");
        // 2. 使用指针遍历数组
        // 循环条件：只要指针 ptr 的地址小于数组末尾的地址 (arr + 5)
        for (int i = 0; i < 5; i++) {
            printf("%d ", *ptr); // *ptr 取出当前指针指向的值
            ptr++;               // ptr++ 让指针指向数组的下一个元素
        }
        printf("\n");

        return 0;
    }
## 指针的加减
对于char数组，地址+1（如0x2c-->0x2d）就是到下一个数组元素了
但对于int数组，每个元素占用四个字节，地址+1则会+4到下一个元素
#### 总结：所谓的“+1”，实则是+ sizeof(type) 
直接访问内容:*(p=1)
#### 指针减指针：得到两指针间距能放下多少个某类型的元素，即 地址距离 / sizeof(type）
## 指针用途：
1. 传回多个值  
2. 返回状态（即0、1也被考虑在结果中）
