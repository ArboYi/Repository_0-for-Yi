# 从后向前排，避免覆盖下个数
## 找/造空位，从空位开始
    #include<stdio.h>

    int main(){
        // 1. 定义数组，长度为10，前9个有值，第10个默认为0（作为预留空间）
        int a[10] = {11, 22, 33, 44, 55, 66, 77, 88, 99};
        int n;
        int i, j;
        int pos = -1; // 用于记录插入位置

        scanf("%d", &n);

        // 2. 寻找插入位置
        // 遍历前9个有效元素，找到第一个比 n 大的数的下标
        for(i = 0; i < 9; i++){
            if(a[i] > n){
                pos = i;
                break; // 找到位置后立即退出循环
            }
        }

        // 3. 处理插入逻辑
        if(pos == -1){
            // 情况A：n 比所有数都大（或者相等），插在最后
            // 此时 pos 保持 -1，我们直接赋值给 a[9]
            a[9] = n;
        } else {
            // 情况B：n 应该插在中间或最前面
            // 需要从最后一个有效元素开始，往后挪一位，直到腾出 pos 位置
            // 注意：要从后往前挪，防止数据覆盖
            for(j = 9; j > pos; j--){
                a[j] = a[j-1];
            }
            // 将 n 放入腾出的位置
            a[pos] = n;
        }

        // 4. 输出结果
        // 现在数组里有10个有效数字
        for(i = 0; i < 10; i++){
            printf("%d ", a[i]);
        }
        
        return 0;
    }
